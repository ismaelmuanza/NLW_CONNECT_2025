import { HcrbyRequest, InMemoryRedisRepositoryInterface } from "./interface/in-memory-redis-repository-interface";


export class InMemoryRedisRepository {

    public store: Record<string, Record<string, number>> = {}

    hincrby (hash: string, field: string, increment: number) {
        if(!this.store[hash]) {
            this.store[hash] = {}
        }

        if(!this.store[hash][field]) {
            this.store[hash][field] = 0
        }

        this.store[hash][field] += increment

        return {
            hash,
            field,
            value: this.store[hash][field]
        }
    }


}
